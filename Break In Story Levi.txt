-- Aguarda os serviços carregarem
local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local LocalPlayer = Players.LocalPlayer

-- Função para localizar o GiveTool
local function localizarGiveTool()
    local caminhos = {
        ReplicatedStorage:FindFirstChild("Events"),
        ReplicatedStorage:FindFirstChild("RemoteEvents")
    }

    for _, caminho in pairs(caminhos) do
        if caminho and caminho:FindFirstChild("GiveTool") then
            return caminho:FindFirstChild("GiveTool")
        end
    end

    return nil
end

-- Tenta carregar a Rayfield
local sucesso, Rayfield = pcall(function()
    return loadstring(game:HttpGet("https://sirius.menu/rayfield"))()
end)

if not sucesso or not Rayfield then
    warn("❌ Falha ao carregar Rayfield!")
    return
end

-- Cria a Janela
local Janela = Rayfield:CreateWindow({
    Name = "🍕 Break In Story Menu - Levi",
    LoadingTitle = "Lulu Engine",
    LoadingSubtitle = "By Levi & Lulu 😈",
    ConfigurationSaving = { Enabled = false },
    Discord = { Enabled = false },
    KeySystem = false
})

-- Aba principal com botões normais
local Principal = Janela:CreateTab("🎁 Itens", 4483362458)

-- Função para criar botão manual
local function criarBotao(nome, item)
    Principal:CreateButton({
        Name = nome,
        Callback = function()
            local GiveTool = localizarGiveTool()
            if GiveTool then
                GiveTool:FireServer(item)
                Rayfield:Notify({
                    Title = "Item entregue!",
                    Content = nome .. " foi adicionado!",
                    Duration = 3
                })
            end
        end,
    })
end

        Name = nome,
        CurrentValue = false,
        Callback = function(ativado)
            task.spawn(function()
                while ativado do
                    local GiveTool = localizarGiveTool()
                    if GiveTool then
                        GiveTool:FireServer(id)
                    end
                    task.wait(1)
                end
            end)
        end
    })
end

-- Aba para inventário
local abaInv = Janela:CreateTab("🧹 Inventário", 4483362458)

-- Alternador: remove todos os itens do inventário em loop
abaInv:CreateToggle({
    Name = "❌ Anti-Itens (Remove tudo)",
    CurrentValue = false,
    Callback = function(ativado)
        task.spawn(function()
            while ativado do
                for _, item in pairs(LocalPlayer.Backpack:GetChildren()) do
                    item:Destroy()
                end
                for _, item in pairs(LocalPlayer.Character:GetChildren()) do
                    if item:IsA("Tool") then
                        item:Destroy()
                    end
                end
                task.wait(0.1)
            end
        end)
    end
})

-- Campo para digitar nome do item e dar automaticamente
abaInv:CreateInput({
    Name = "🔎 Procurar e Pegar Item",
    PlaceholderText = "Escreva o nome exato do item",
    RemoveTextAfterFocusLost = false,
    Callback = function(valor)
        local GiveTool = localizarGiveTool()
        if GiveTool then
            GiveTool:FireServer(valor)
            Rayfield:Notify({
                Title = "Item procurado!",
                Content = "Tentando dar: " .. valor,
                Duration = 3
            })
        end
    end
})
