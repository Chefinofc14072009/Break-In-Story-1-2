-- ======================
-- LOCALIZAR GIVETOOL
-- ======================
local function localizarGiveTool()
    local caminhos = {
        ReplicatedStorage:FindFirstChild("Events"),
        ReplicatedStorage:FindFirstChild("RemoteEvents")
    }

    for _, pasta in pairs(caminhos) do
        if pasta and pasta:FindFirstChild("GiveTool") then
            return pasta:FindFirstChild("GiveTool")
        end
    end
    return nil
end

-- ABA ENERGIA ⚡
-- ======================
local Energia = Janela:CreateTab("⚡ Energia", 4483362458)

local function treinar(stat)
    ReplicatedStorage
        :WaitForChild("Events")
        :WaitForChild("RainbowWhatStat")
        :FireServer(stat)
end

Energia:CreateButton({
    Name = "Treinar força 💪",
    Callback = function()
        treinar("Strength")
    end
})

Energia:CreateButton({
    Name = "Treinar velocidade 🧬",
    Callback = function()
        treinar("Speed")
    end
})

Energia:CreateButton({
    Name = "Treino máximo 🔥",
    Callback = function()
        for i = 1, 5 do
            treinar("Strength")
            treinar("Speed")
        end
    end
})

-- ABA INVENTÁRIO (ÚLTIMA)
-- ======================
local Inv = Janela:CreateTab("🧹 Inventário", 4483362458)

Inv:CreateToggle({
    Name = "❌ Anti-Itens (Remove tudo)",
    CurrentValue = false,
    Callback = function(v)
        task.spawn(function()
            while v do
                for _, t in pairs(LocalPlayer.Backpack:GetChildren()) do
                    t:Destroy()
                end
                if LocalPlayer.Character then
                    for _, t in pairs(LocalPlayer.Character:GetChildren()) do
                        if t:IsA("Tool") then
                            t:Destroy()
                        end
                    end
                end
                task.wait(0.1)
            end
        end)
    end
})

Inv:CreateInput({
    Name = "🔎 Procurar e pegar item",
    PlaceholderText = "Nome exato do item",
    RemoveTextAfterFocusLost = false,
    Callback = function(valor)
        local GiveTool = localizarGiveTool()
        if GiveTool then
            GiveTool:FireServer(valor)
        end
    end
})
